<script>
// Not every Julia version has its own docs: some tags were never deployed, and there is no
// `vX.Y` until X.Y is released. For a missing version, go to the same page in the nearest
// version that `versions.js` lists. Listed versions are never redirected, so this cannot loop.
function nearestVersion(requested, versions, stable) {
  if (requested === "stable") return stable;
  if (requested === "dev" || requested === "latest") {
    var dev = versions.filter(function (v) { return /-dev$/.test(v); });
    return dev.length ? dev[dev.length - 1] : stable;
  }
  if (versions.indexOf(requested) >= 0) return null;
  var m = /^v(\d+)\.(\d+)(\.|-|$)/.exec(requested);
  if (!m) return null;
  var minor = "v" + m[1] + "." + m[2];
  var same = versions.filter(function (v) {
    return v === minor || v.indexOf(minor + ".") === 0 || v.indexOf(minor + "-") === 0;
  });
  // the release first, then its prereleases, then its development docs
  same.sort(function (a, b) { return (a !== minor) - (b !== minor); });
  return same.length ? same[0] : stable;
}

(function () {
  var m = /^\/en\/([^\/]+)(\/.*)?$/.exec(window.location.pathname);
  if (!m) return;
  var requested = m[1], page = m[2] || "/";
  var script = document.createElement("script");
  script.src = "/en/versions.js";
  script.onload = function () {
    if (typeof DOC_VERSIONS === "undefined") return;
    var target = nearestVersion(requested, DOC_VERSIONS, DOCUMENTER_STABLE);
    if (!target || target === requested) return;
    var url = "/en/" + target + page + window.location.search + window.location.hash;
    var box = document.getElementById("version-redirect");
    var count = document.getElementById("version-redirect-count");
    var seconds = 5;
    document.getElementById("version-redirect-text").textContent =
      "There are no docs for Julia " + requested + " at this address. " +
      "The nearest are the docs for " + target + ".";
    document.getElementById("version-redirect-go").href = url;
    function tick() {
      count.textContent = "Going to the " + target + " docs in " + seconds + " s…";
      if (seconds-- === 0) window.location.replace(url);
    }
    var timer = setInterval(tick, 1000);
    tick();
    document.getElementById("version-redirect-stay").onclick = function (e) {
      e.preventDefault();
      clearInterval(timer);
      count.textContent = "Staying on this page.";
      this.style.display = "none";
    };
    box.hidden = false;
  };
  document.head.appendChild(script);
})();
</script>
